# Invisible storyline initialization mission.
# Gives the real Babylon 5 start the same initially explored map as Testbed.

mission "b5 story known space initializer"
	name "B5 Story Known Space Initializer"
	landing
	invisible
	non-blocking
	repeat
	to offer
		has "b5 story start"
		not "b5 story known space initialized"
	source "Babylon 5"
	on offer
		event "b5 testbed known space" 0
		set "b5 story known space initialized"
		fail

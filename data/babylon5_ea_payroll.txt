# Earth Alliance officer payroll.
# Endless Sky 0.11.2 parser-safe implementation.
#
# The tracker counts in-game days while the Babylon 5 story is active.
# Once seven days have accrued, the invisible payout mission deposits
# 1,000 credits on the player's next landing and subtracts seven days.

mission "b5 earth alliance officer payroll tracker"
	name "Earth Alliance Officer Payroll Tracker"
	landing
	invisible
	non-blocking
	to offer
		has "b5 story start"
	on offer
		set "b5 ea payroll active"
	on daily
		"b5 ea payroll days" ++

mission "b5 earth alliance officer payroll payout"
	name "Earth Alliance Officer Payroll"
	landing
	invisible
	non-blocking
	repeat
	to offer
		has "b5 story start"
		"b5 ea payroll days" >= 7
	on offer
		payment 1000
		"b5 ea payroll days" -= 7
		fail
